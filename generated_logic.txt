#include <iostream>
using namespace std;

int main() {
    int num1, num2;
    char op;

    cout << "Enter first number: ";
    cin >> num1;

    cout << "Enter operator (+, -, *, /): ";
    cin >> op;

    cout << "Enter second number: ";
    cin >> num2;

    switch (op) {
    case '+':
        cout << "The result is: " << (num1 + num2) << endl;
        break;
    case '-':
        cout << "The result is: " << (num1 - num2) << endl;
        break;
    case '*':
        cout << "The result is: " << (num1 * num2) << endl;
        break;
    case '/':
        if (num2 != 0)
            cout << "The result is: " << (num1 / num2) << endl;
        else
            cout << "Error! Division by zero." << endl;
        break;
    default:
        cout << "Error! Invalid operator." << endl;
    }

    return 0;
}
